<LessonStep number={2} title="Apply the versioned NestJS repair" file="drafts/nestjs/labs/async-module-configuration/solution.ts" expected="Dynamic modules should resolve and validate async configuration before exposing clients to consumers.">

```ts
import { DynamicModule, Module } from '@nestjs/common';
export const API_URL = Symbol('API_URL');
@Module({})
export class ApiModule {
  static forRootAsync(): DynamicModule {
    return {
      module: ApiModule,
      providers: [{
        provide: API_URL,
        useFactory: async () => {
          const raw = process.env.API_URL ?? 'https://example.org';
          const endpoint = new URL(raw);
          if (endpoint.protocol !== 'https:') throw new Error('HTTPS API URL required');
          return endpoint.toString();
        },
      }],
      exports: [API_URL],
    };
  }
}
```

**Why it works:** Dynamic modules should resolve and validate async configuration before exposing clients to consumers. Never log raw credentials, and do not create connections before configuration is validated.

</LessonStep>
