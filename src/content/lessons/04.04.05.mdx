## The problem we're solving

A multi-package workspace builds modules in an arbitrary order and sometimes consumes outdated type declarations. This is a local repository/package-development lesson. A passing fixture alone is not a published npm release and does not create a Vercel deployment.

## Build the mental model

Project references express a dependency graph, enabling TypeScript build mode to compile referenced projects in order. composite and declaration settings are generally required on referenced projects. Cycles cannot be resolved by topological build order; fail explicitly and review tsconfig paths.

**Predict before running:** Referenced libraries build before dependents, and cycles or unknown dependencies fail deterministically.

type Project={name:string;deps:string[]};export function solve(projects:readonly Project[]):string[]{const byName=new Map(projects.map(p=>[p.name,p]));if(byName.size!==projects.length)throw new Error('Duplicate project');const ordered:string[]=[],active=new Set<string>(),complete=new Set<string>();function visit(name:string){if(complete.has(name))return;if(active.has(name))throw new Error('Cycle');const project=byName.get(name);if(!project)throw new Error('Unknown dependency: '+name);active.add(name);for(const dep of project.deps)visit(dep);active.delete(name);complete.add(name);ordered.push(name);}for(const project of projects)visit(project.name);return ordered;}
