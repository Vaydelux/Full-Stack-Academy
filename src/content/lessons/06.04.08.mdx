## The problem we're solving

A comment composer needs immediate feedback and a consistent record list while an asynchronous save is simulated locally. The starter's reconciliation duplicates an existing comment whenever the same record is committed twice.

## Build the mental model

Coordinate optimistic insertion, eventual reconciliation and duplicate protection without claiming local simulation is server persistence. Discard or reconcile the temporary pending item and show a meaningful error, rather than presenting it as saved. The React 19 APIs used here have distinct contracts: context passes values, transitions schedule nonurgent updates, deferred values let rendering lag, and optimistic state is a temporary view during an Action. These lessons use local data, not backend persistence.

**Predict before you run:** Committing the same identifier twice preserves one record, and an optimistic item cannot overwrite an existing committed entry.

import {startTransition,useOptimistic,useRef,useState,type FormEvent} from 'react';
export type Entry={id:string;text:string;pending:boolean};
export function reconcile(current:readonly Entry[],item:Entry):Entry[]{
 if(current.some(existing=>existing.id===item.id))return current.map(existing=>existing.id===item.id?item:existing);
 return [...current,item];
}
export function CommentsCapstone(){
 const [saved,setSaved]=useState<Entry[]>([]);const [shown,addPending]=useOptimistic(saved,(items:Entry[],item:Entry)=>[...items,item]);
 const [draft,setDraft]=useState('');const sequence=useRef(0);
 function submit(e:FormEvent<HTMLFormElement>){e.preventDefault();const text=draft.trim();if(!text)return;

   startTransition(()=>setSaved(curr=>reconcile(curr,{...item,pending:false})));});setDraft('');
 }
 return <section><h2>Comments</h2><form onSubmit={submit}><label htmlFor="capstone-comment">Message</label><input id="capstone-comment" value={draft} onChange={e=>setDraft(e.target.value)}/>

}
